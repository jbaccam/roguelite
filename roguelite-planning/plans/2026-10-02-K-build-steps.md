```python
"""Anchored hunks for scripts several sessions edit at once (memory: shared-script edits as hunks).

A plan's hunk file defines HUNKS (and optionally MODULES) and calls run():
    python <plan>_hunks.py            apply to the repo files (hunks already there are skipped)
    python <plan>_hunks.py --check    report only, change nothing
    python <plan>_hunks.py --stage    also stage HEAD + these hunks (git index) for each shared file,
                                      so a commit never takes other sessions' uncommitted edits
    python <plan>_hunks.py --json     also write the Studio sync manifest for tools/SyncPlan.luau

Hunk: (repo file, Studio dot path or None, anchor, where, text). where is 'after' / 'before' the
anchor line, 'replace' the anchor line, or 'block': anchor is (first line, last line) and the whole
block is replaced. Lines match with a trailing \\r ignored. A hunk is "present" when its text
already appears in the file.
"""
import json
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent   # studio-prototype


def apply(text, anchor, where, ins):
    if ins in text.replace("\r\n", "\n"):
        return text, "present"
    lines = text.split("\n")

    def find(a):
        return [i for i, l in enumerate(lines) if l.rstrip("\r") == a]

    def fresh(i):
        crlf = lines[i].endswith("\r")
        return [l + ("\r" if crlf else "") for l in ins.split("\n")]

    if where == "block":
        a, b = find(anchor[0]), find(anchor[1])
        if len(a) != 1 or len(b) != 1 or b[0] < a[0]:
            return text, f"block anchors found {len(a)}/{len(b)} times"
        lines[a[0]:b[0] + 1] = fresh(a[0])
        return "\n".join(lines), "applied"
    hits = find(anchor)
    if len(hits) != 1:
        return text, f"anchor found {len(hits)} times"
    i = hits[0]
    new = fresh(i)
    if where == "replace":
        lines[i:i + 1] = new
    elif where == "after":
        lines[i + 1:i + 1] = new
    elif where == "before":
        lines[i:i] = new
    else:
        return text, f"unknown where {where}"
    return "\n".join(lines), "applied"


def stage_on_head(hunks):
    repo = subprocess.run(["git", "rev-parse", "--show-toplevel"], cwd=ROOT, capture_output=True, text=True, check=True).stdout.strip()
    prefix = subprocess.run(["git", "rev-parse", "--show-prefix"], cwd=ROOT, capture_output=True, text=True, check=True).stdout.strip()
    for path in sorted({h[0] for h in hunks}):
        gp = prefix + path
        text = subprocess.run(["git", "show", "HEAD:" + gp], cwd=repo, capture_output=True, check=True).stdout.decode("utf-8")
        for p, _, anchor, where, ins in hunks:
            if p == path:
                text, status = apply(text, anchor, where, ins)
                assert status in ("applied", "present"), f"{path}: {status}"
        sha = subprocess.run(["git", "hash-object", "-w", "--stdin"], cwd=repo, input=text.encode("utf-8"), capture_output=True, check=True).stdout.decode().strip()
        subprocess.run(["git", "update-index", "--cacheinfo", f"100644,{sha},{gp}"], cwd=repo, check=True)
        print("staged HEAD + hunks:", gp)


def run(hunks, modules=(), manifest=None, base=None):
    check = "--check" in sys.argv
    files, bad = {}, 0
    for path, studio, anchor, where, ins in hunks:
        p = ROOT / path
        text = files.get(path) or p.read_bytes().decode("utf-8")
        new, status = apply(text, anchor, where, ins)
        files[path] = new
        bad += status not in ("applied", "present")
        print(f"{status:30s} {path}: {ins.splitlines()[0][:70]}")
    if bad:
        print(f"{bad} hunk(s) did not apply; nothing written")
        sys.exit(1)
    if not check:
        for path, text in files.items():
            p = ROOT / path
            if text != p.read_bytes().decode("utf-8"):
                p.write_bytes(text.encode("utf-8"))
    if "--stage" in sys.argv:
        stage_on_head(hunks)
    if "--json" in sys.argv and manifest:
        out = {
            "modules": [dict(m, base=base) for m in modules],
            "hunks": [{"repo": p, "studio": s, "anchor": a, "where": w, "text": t} for p, s, a, w, t in hunks if s],
        }
        (ROOT / manifest).write_text(json.dumps(out, indent=1, ensure_ascii=False), encoding="utf-8")
        print("wrote", manifest)
```

```python
"""Local server for the Studio tools (plan K). Binds 127.0.0.1 only.
    python roguelite-planning/studio-prototype/tools/dev_server.py [port]      (default 8934)
GET  /<path>                  files under roguelite-planning/ (e.g. /studio-prototype/combat/QuestConfig.luau)
GET  /git/<rev>/<path>        a roguelite-planning/ file as committed at <rev> (SyncPlan's guard)

}


def asset_map():
    m = dict(THEME)
    for f in (PROTO / "ui" / "assets").rglob("*asset-ids.json"):
        def add(value, key, folder=f.parent):
            hit = re.search(r"rbxassetid://(\d+)", value)
            png = folder / f"{key}.png"
            if hit and png.exists():
                m.setdefault(hit.group(1), png.relative_to(PLANNING).as_posix())

        def walk(node, key=None):
            if isinstance(node, dict):
                if key and isinstance(node.get("asset"), str):
                    add(node["asset"], key)
                for k, v in node.items():
                    walk(v, k)
            elif isinstance(node, str) and key:
                add(node, key)

        walk(json.loads(f.read_text(encoding="utf-8")))
    return m


class Handler(SimpleHTTPRequestHandler):
    def end_headers(self):
        self.send_header("Cache-Control", "no-store")
        super().end_headers()

    def send_bytes(self, data, ctype):
        self.send_response(200)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        path = self.path.split("?")[0]
        if path == "/assetmap.json":
            return self.send_bytes(json.dumps(asset_map()).encode(), "application/json")
        if path.startswith("/git/"):
            _, _, rev, rest = path.split("/", 3)
            r = subprocess.run(["git", "show", f"{rev}:roguelite-planning/{rest}"], cwd=PLANNING, capture_output=True)
            if r.returncode:
                return self.send_error(404, r.stderr.decode(errors="replace")[:200])
            return self.send_bytes(r.stdout.replace(b"\r\n", b"\n"), "text/plain; charset=utf-8")
        if path.startswith("/dump/"):
            f = DUMPS / Path(path).name
            if not f.exists():
                return self.send_error(404)
            return self.send_bytes(f.read_bytes(), "application/json")
        return super().do_GET()

    def do_POST(self):
        path = self.path.split("?")[0]
        if not path.startswith("/dump/"):
            return self.send_error(404)
        name = re.sub(r"[^\w-]", "", Path(path).name) or "preview"
        body = self.rfile.read(int(self.headers.get("Content-Length", 0)))
        DUMPS.mkdir(exist_ok=True)
        (DUMPS / f"{name}.json").write_bytes(body)
        self.send_bytes(b"ok", "text/plain")


if __name__ == "__main__":
    port = int(sys.argv[1]) if len(sys.argv) > 1 else 8934
    print(f"serving {PLANNING} on http://127.0.0.1:{port}/")
    ThreadingHTTPServer(("127.0.0.1", port), partial(Handler, directory=str(PLANNING))).serve_forever()
```

```lua
--!nonstrict
-- Fresh copies of repo modules in Studio Edit, for checks and UI previews without Play (plan K).
-- The plugin VM caches require() across execute_luau calls, so Studio's ModuleScripts can come back
-- stale. Names in PATHS load from the repo through tools/dev_server.py instead, and their requires
-- of each other (by instance name) resolve to the same fresh copies; everything else is required
-- from Studio as usual. Usage (execute_luau):
--   local HS=game:GetService('HttpService')
--   local base='http://127.0.0.1:8934/studio-prototype/'
--   local Fresh=loadstring(HS:GetAsync(base..'tools/FreshRequire.luau?t='..os.clock(),true))()(base)
--   local QuestsUI=Fresh.require('QuestsUI')
return function(base)
	local HS = game:GetService("HttpService")
	local PATHS = {
		QuestConfig = "combat/QuestConfig.luau",
		QuestService = "combat/QuestService.luau",
		QuestTests = "combat/QuestTests.luau",
		CharacterStats = "combat/CharacterStats.luau",
		QuestsUI = "ui/QuestsUI.luau",
		StatPlates = "ui/StatPlates.luau",
		UITheme = "ui/UITheme.luau",
		StoreFX = "ui/StoreFX.luau",
		UILayoutAudit = "ui/UILayoutAudit.luau",
		RunSetupRules = "lobby/RunSetupRules.luau",
	}
	-- Where each module lives in Studio: its fresh copy sees that instance as `script`, so
	-- `script.Parent.X` requires still find their (Studio) siblings.
	local STUDIO = {
		QuestConfig = "ReplicatedStorage.RogueliteCombat.QuestConfig",
		QuestService = "ServerScriptService.QuestService",
		CharacterStats = "ReplicatedStorage.RogueliteCombat.CharacterStats",
		QuestsUI = "ReplicatedStorage.QuestsUI",
		StatPlates = "ReplicatedStorage.StatPlates",
		UITheme = "ReplicatedStorage.UITheme",
		StoreFX = "ReplicatedStorage.StoreFX",
		UILayoutAudit = "ReplicatedStorage.UILayoutAudit",
		RunSetupRules = "ReplicatedStorage.RunSetupRules",
	}
	local function resolve(path)
		local parts = string.split(path or "", ".")
		local ok, inst = pcall(game.GetService, game, parts[1])
		inst = ok and inst or nil
		for i = 2, #parts do
			inst = inst and inst:FindFirstChild(parts[i])
		end
		return inst
	end
	local F = { cache = {} }
	function F.source(path)
		return (HS:GetAsync(base .. path .. "?t=" .. os.clock(), true):gsub("\r\n", "\n"))
	end
	function F.compiles(path)
		local fn, err = loadstring(F.source(path), "=" .. path)
		return fn ~= nil, err
	end
	function F.require(target)
		local name = typeof(target) == "Instance" and target.Name or target
		local path = PATHS[name]
		if not path then
			return require(target)
		end
		if F.cache[name] == nil then
			local fn, err = loadstring(F.source(path), "=" .. path)
			assert(fn, tostring(err))
			local here = typeof(target) == "Instance" and target or resolve(STUDIO[name])
			setfenv(fn, setmetatable({ require = F.require, script = here }, { __index = getfenv(0) }))
			F.cache[name] = fn()
		end
		return F.cache[name]
	end
	return F
end
```

```lua
--!nonstrict
-- A GUI tree as JSON for tools/gui_view.html (plan K): one entry per visible GuiObject in draw
-- order, positions relative to `root`, with what the viewer needs to redraw it (colours, corners,
-- strokes, gradients, images and 9-slices, text, clipping). Usage (execute_luau):
--   local dump=loadstring(Fresh.source('tools/GuiDump.luau'))()
--   HS:PostAsync('http://127.0.0.1:8934/dump/quests-daily',HS:JSONEncode(dump(root)))
return function(root)
	local out = {}
	local origin = root.AbsolutePosition
	local function rgb(c)
		return { math.round(c.R * 255), math.round(c.G * 255), math.round(c.B * 255) }
	end
	local function walk(g, clip)
		if not g:IsA("GuiObject") or not g.Visible then
			return
		end
		local p, s = g.AbsolutePosition - origin, g.AbsoluteSize
		local e = { n = g.Name, x = p.X, y = p.Y, w = s.X, h = s.Y, r = g.AbsoluteRotation, cl = clip }
		if g.BackgroundTransparency < 1 and not g:IsA("ScrollingFrame") then
			e.bg = rgb(g.BackgroundColor3)
			e.bt = g.BackgroundTransparency
		end
		local isText = g:IsA("TextLabel") or g:IsA("TextButton") or g:IsA("TextBox")
		for _, c in g:GetChildren() do
			if c:IsA("UICorner") then
				e.cr = c.CornerRadius.Scale * math.min(s.X, s.Y) + c.CornerRadius.Offset
			elseif c:IsA("UIStroke") and c.Enabled then
				local onText = isText and c.ApplyStrokeMode == Enum.ApplyStrokeMode.Contextual
				e[onText and "ts" or "bs"] = { rgb(c.Color), c.Thickness, c.Transparency }
			elseif c:IsA("UIGradient") and c.Enabled then
				local k = c.Color.Keypoints
				e.gr = { rgb(k[1].Value), rgb(k[#k].Value), c.Rotation, #k > 2 and rgb(k[2].Value) or nil }
				local t = c.Transparency.Keypoints
				e.gt = { t[1].Value, t[#t].Value }
			end
		end
		if (g:IsA("ImageLabel") or g:IsA("ImageButton")) and g.Image ~= "" and g.ImageTransparency < 1 then
			e.img = g.Image
			e.ic = rgb(g.ImageColor3)
			e.it = g.ImageTransparency
			if g.ScaleType == Enum.ScaleType.Slice then
				e.sl = { g.SliceCenter.Min.X, g.SliceScale }
			end
		end
		if isText and g.Text ~= "" and g.TextTransparency < 1 then
			e.t = g.ContentText
			e.tc = rgb(g.TextColor3)
			e.tt = g.TextTransparency
			e.tb = { g.TextBounds.X, g.TextBounds.Y }
			e.xa = g.TextXAlignment.Name
			e.ya = g.TextYAlignment.Name
			e.f = g.Font.Name
			if g.TextStrokeTransparency < 1 then
				e.tst = g.TextStrokeTransparency
			end
		end
		table.insert(out, e)
		if g.ClipsDescendants then
			local x0, y0, x1, y1 = p.X, p.Y, p.X + s.X, p.Y + s.Y
			if clip then
				x0, y0 = math.max(x0, clip[1]), math.max(y0, clip[2])
				x1, y1 = math.min(x1, clip[1] + clip[3]), math.min(y1, clip[2] + clip[4])
			end
			clip = { x0, y0, math.max(0, x1 - x0), math.max(0, y1 - y0) }
		end
		-- Sibling ZIndexBehavior: higher ZIndex draws later; ties keep creation order.
		local kids = {}
		for i, c in g:GetChildren() do
			if c:IsA("GuiObject") then
				table.insert(kids, { c, i })
			end
		end
		table.sort(kids, function(a, b)
			if a[1].ZIndex ~= b[1].ZIndex then
```

```lua
--!nonstrict
-- Edit-mode Studio sync for a plan's changes (plan K; generalizes lobby/SyncEggMerchant.luau). With
-- tools/dev_server.py running, call from execute_luau:
--   local HS=game:GetService('HttpService')
--   local base='http://127.0.0.1:8934/'
--   return loadstring(HS:GetAsync(base..'studio-prototype/tools/SyncPlan.luau?t='..os.clock(),true))()(base,'studio-prototype/combat/plan-k-sync.json',true)
-- The manifest (a plan's *_hunks.py --json) has:
--   modules: {path, parent (dot path), name, base (git rev), sandboxLike (dot path, optional)}:
--     whole modules the plan owns. One is written only if Studio's Source is still what `base`
--     committed (or already the new source), so an edit made in Studio since is never overwritten.
--     A new module copies Sandboxed and Capabilities from sandboxLike.
--   hunks: {repo, studio, anchor, where, text}: anchored edits to shared scripts ('after', 'before',
--     'replace', or 'block' with anchor {first, last}).
-- All or nothing: every module and patched script compiles, every anchor matches once and every
-- guard passes before anything is written. dryRun reports without writing. Returns a report.
return function(base, manifestPath, dryRun)
	local HS = game:GetService("HttpService")
	local History = game:GetService("ChangeHistoryService")
	local function get(path)
		return (HS:GetAsync(base .. path .. "?t=" .. os.clock(), true):gsub("\r\n", "\n"))
	end
	local function compiles(name, src)
		local fn, err = loadstring(src, "=" .. name)
		return fn ~= nil, err
	end
	local function resolve(path)
		local parts = string.split(path, ".")
		local inst = game:GetService(parts[1])
		for i = 2, #parts do
			inst = inst and inst:FindFirstChild(parts[i])
		end
		return inst
	end
	local manifest = HS:JSONDecode(get(manifestPath))
	local report = {}

	local modules = {}
	for _, m in manifest.modules or {} do
		local parent = resolve(m.parent)
		if not parent then
			return "ABORT: " .. m.parent .. " not found"
		end
		local source = get("studio-prototype/" .. m.path)
		local ok, err = compiles(m.name, source)
		if not ok then
			return "ABORT: " .. m.path .. " does not compile: " .. tostring(err)
		end
		local inst = parent:FindFirstChild(m.name)
		if inst then
			local now = inst.Source:gsub("\r\n", "\n")
			if now ~= source then
				local okBase, was = pcall(get, "git/" .. m.base .. "/studio-prototype/" .. m.path)
				if not okBase or now ~= was then
					return "ABORT: Studio's " .. m.parent .. "." .. m.name .. " is not the " .. m.base .. " version (edited in Studio, or Studio lags); merge it by hand"
				end
			end
		end
		table.insert(modules, { m = m, parent = parent, inst = inst, source = source })
	end

	local targets, order = {}, {}
	for _, h in manifest.hunks or {} do
		local t = targets[h.studio]
		if not t then
			local inst = resolve(h.studio)
			if not inst then
				return "ABORT: " .. h.studio .. " not found"
			end
			t = { inst = inst, source = inst.Source:gsub("\r\n", "\n"), added = 0, present = 0 }
			targets[h.studio] = t
			table.insert(order, h.studio)
		end
		if string.find(t.source, h.text, 1, true) then
			t.present += 1
		else
			local lines = string.split(t.source, "\n")
			local function find(a)
				local at, hits = nil, 0
				for i, l in lines do
					if l == a then
						at, hits = i, hits + 1
					end
				end
				return at, hits
			end
			local ins = string.split(h.text, "\n")
			if h.where == "block" then
				local a, na = find(h.anchor[1])
				local z, nz = find(h.anchor[2])
				if na ~= 1 or nz ~= 1 or z < a then
					return "ABORT: block anchors in " .. h.studio .. " found " .. na .. "/" .. nz .. " times: " .. h.anchor[1]
				end
				for _ = a, z do
					table.remove(lines, a)
				end
				for k, l in ins do
					table.insert(lines, a + k - 1, l)
				end
			else
				local at, hits = find(h.anchor)
				if hits ~= 1 then
					return "ABORT: anchor found " .. hits .. " times in " .. h.studio .. ": " .. h.anchor
				end
				if h.where == "replace" then
					table.remove(lines, at)
				end
				local pos = h.where == "after" and at + 1 or at
				for k, l in ins do
					table.insert(lines, pos + k - 1, l)
				end
			end
			t.source = table.concat(lines, "\n")
			t.added += 1
		end
	end
	for _, path in order do
		local ok, err = compiles(path, targets[path].source)
		if not ok then
			return "ABORT: patched " .. path .. " does not compile: " .. tostring(err)
		end
	end

	if dryRun then
		for _, e in modules do
			table.insert(report, e.m.name .. ": " .. (not e.inst and "would create" or e.inst.Source:gsub("\r\n", "\n") == e.source and "unchanged" or "would update"))
		end
		for _, path in order do
			table.insert(report, path .. ": " .. targets[path].added .. " to add, " .. targets[path].present .. " present")
		end
		return "DRY RUN\n" .. table.concat(report, "\n")
	end

	History:SetWaypoint("Before " .. manifestPath)
	for _, e in modules do
		local inst, created = e.inst, false
		if not inst then
			inst = Instance.new("ModuleScript")
			inst.Name = e.m.name
			created = true
			local like = e.m.sandboxLike and resolve(e.m.sandboxLike)
			if like then
				pcall(function()
					inst.Sandboxed = like.Sandboxed
				end)
				pcall(function()
					inst.Capabilities = like.Capabilities
				end)
			end
		end
		if inst.Source:gsub("\r\n", "\n") ~= e.source then
			inst.Source = e.source
			table.insert(report, e.m.name .. (created and ": created" or ": updated"))
		else
			table.insert(report, e.m.name .. ": unchanged")
		end
		inst.Parent = e.parent
	end
	for _, path in order do
		local t = targets[path]
		if t.added > 0 then
			t.inst.Source = t.source
		end
		table.insert(report, path .. ": " .. t.added .. " hunks added, " .. t.present .. " already there")
	end
	History:SetWaypoint("After " .. manifestPath)
	return table.concat(report, "\n")
end
```
